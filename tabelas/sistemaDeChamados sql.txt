create database sistemadechamados;

use sistemadechamados;

-- 1. CRIAÇÃO DA TABELA DE CLIENTES
CREATE TABLE `clientes` (
  `id_cliente` int(11) NOT NULL AUTO_INCREMENT,
  `nome_empresa` varchar(255) NOT NULL,
  `localizacao` varchar(255) DEFAULT NULL,
  `contato_principal` varchar(100) DEFAULT NULL,
  `num_celular` varchar(20) DEFAULT NULL,
  `email_contato` varchar(100) DEFAULT NULL,
  `status_cliente` varchar(20) DEFAULT 'Ativo',
  PRIMARY KEY (`id_cliente`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2. CRIAÇÃO DA TABELA DE USUÁRIOS (COM ID EM MINÚSCULO)
CREATE TABLE `usuarios` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `nome` varchar(255) NOT NULL,
  `email` varchar(100) NOT NULL,
  `senha` varchar(255) NOT NULL,
  `num_celular` varchar(20) DEFAULT NULL,
  `perfil` varchar(20) NOT NULL,
  `id_cliente` int(11) DEFAULT NULL,
  `status` varchar(20) DEFAULT 'Ativo',
  `criado_em` datetime DEFAULT current_timestamp(),
  `atualizado_em` datetime DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `email` (`email`),
  KEY `id_cliente` (`id_cliente`),
  CONSTRAINT `usuarios_ibfk_1` FOREIGN KEY (`id_cliente`) REFERENCES `clientes` (`id_cliente`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 3. CRIAÇÃO DA TABELA DE CHAMADOS
CREATE TABLE `chamados` (
  `id_chamado` int(11) NOT NULL AUTO_INCREMENT,
  `id_cliente` int(11) NOT NULL,
  `id_usuario` int(11) NOT NULL,
  `id_tecnico_atribuido` int(11) DEFAULT NULL,
  `data_abertura` datetime DEFAULT current_timestamp(),
  `data_fechamento` datetime DEFAULT NULL,
  `status` varchar(20) DEFAULT 'Novo',
  `prioridade` varchar(20) DEFAULT 'Media',
  `descricao_solicitacao` text NOT NULL,
  `solucao` text DEFAULT NULL,
  `origem` varchar(20) DEFAULT 'Portal',
  PRIMARY KEY (`id_chamado`),
  KEY `id_cliente` (`id_cliente`),
  KEY `id_usuario` (`id_usuario`),
  KEY `id_tecnico_atribuido` (`id_tecnico_atribuido`),
  CONSTRAINT `chamados_ibfk_1` FOREIGN KEY (`id_cliente`) REFERENCES `clientes` (`id_cliente`) ON DELETE CASCADE,
  CONSTRAINT `chamados_ibfk_2` FOREIGN KEY (`id_usuario`) REFERENCES `usuarios` (`id`) ON DELETE CASCADE,
  CONSTRAINT `chamados_ibfk_3` FOREIGN KEY (`id_tecnico_atribuido`) REFERENCES `usuarios` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
